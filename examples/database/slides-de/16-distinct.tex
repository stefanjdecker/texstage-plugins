\begin{quiz}[id=distinct-keyword,type=text,time=40]{Welches SQL-Schlüsselwort entfernt doppelte Ergebniszeilen?}
\accept{DISTINCT}
\explain{SELECT DISTINCT entfernt Duplikate aus den gewählten Ergebniszeilen.}
\note{Groß- und Kleinschreibung werden bei der Antwort normalisiert. DISTINCT garantiert nicht eine Zeile je Mitglied, wenn weitere ausgewählte Werte abweichen.}
\end{quiz}
