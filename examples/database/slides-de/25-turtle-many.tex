\begin{frame}[t]{Turtle-Kurzschreibweisen verwenden}
\small Verschiedene Turtle-Texte können denselben Graphen beschreiben.\par\vspace{3mm}
\begin{activity}[id=turtle-many,input=code,engine=turtle,seed=data/library.ttl,
reference=answers/turtle-many.ttl,open=manual,limit=5,height=.61\textheight]
\prompt{Ada hat Database Basics und Practical SQL ausgeliehen, einschließlich früherer Ausleihen. Nutzen Sie \texttt{:ada :hasBorrowed} mit \texttt{:b101} und \texttt{:b103}. Schreiben Sie genau zwei Tripel; eine Komma-Liste ist möglich.}
\end{activity}
\note{Eine Komma-Liste und zwei einzelne Aussagen sind gleichwertig. Identische wiederholte Tripel ergänzen keine Graphfakten. hasBorrowed umfasst ausdrücklich Adas frühere zurückgegebene Ausleihe.}
\end{frame}
