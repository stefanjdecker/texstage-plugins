\begin{quiz}[id=distinct-keyword,type=text,time=40]{Which SQL keyword removes duplicate result rows?}
\accept{DISTINCT}
\explain{SELECT DISTINCT removes duplicates from the selected result rows.}
\note{The built-in normalization ignores case. DISTINCT is not one row per member when other selected values differ.}
\end{quiz}
